/* ========== 新增：嵌套标签与注释清理 ========== */
function cleanTags(text){
var prev;
/* 1. 移除HTML注释（匹配 <!-- 任意内容 -->） */
text=text.replace(/<!--[\s\S]*?-->/g,'');

/* 2. 循环移除所有英文标签对及其内容（从最内层开始，逐层向外） */
do{
prev=text;
text=text.replace(/<([a-z_][a-z0-9_]*)[^>]*>[^<]*<\/\1>/gi,'');
}while(text!==prev);

/* 3. 移除自闭合标签 <xxx/> */
text=text.replace(/<[a-z_][a-z0-9_]*[^>]*\/>/gi,'');

/* 4. 移除残留的单独开/闭标签 */
text=text.replace(/<\/?[a-z_][a-z0-9_]*[^>]*>/gi,'');

/* 5. 清理多余空行 */
text=text.replace(/\n{3,}/g,'\n\n');

return text.trim();
}

    h+='<span class="ring-lb2">'+(got?esc(rd.年限||''):'—')+'</span></div>';
  }
  return h+'</div></div>';
}

function refreshModal(){

  $('.md-scroll').css('opacity',0);
  setTimeout(function(){

    else if(curModal.type==='m') showMaleModal(curModal.name,g('NPC列表.'+curModal.name,{}));

    else if(curModal.type==='bone') rebuildBoneModal(curModal.bt,curModal.bk);

    $('.md-scroll').css('opacity',1);
  },20);
}

async function saveMvu(ep,val){
  try{
    await waitGlobalInitialized('Mvu');
    var msgId='latest';try{if(typeof getCurrentMessageId==='function') msgId=getCurrentMessageId();}catch(e){}
    var data=Mvu.getMvuData({type:'message',message_id:msgId});
    _.set(data,'stat_data.'+ep,val);
    await Mvu.replaceMvuData(data,{type:'message',message_id:msgId});
    return true;
  }catch(err){console.error('Mvu save failed',err);return false;}
}

async function deleteNpcMvu(type, name){
  try{
    await waitGlobalInitialized('Mvu');
    var msgId='latest';try{if(typeof getCurrentMessageId==='function') msgId=getCurrentMessageId();}catch(e){}
    var data=Mvu.getMvuData({type:'message',message_id:msgId});
    var targetObj=_.get(data,'stat_data.'+type);
    if(targetObj && targetObj[name]){
      delete targetObj[name];
      await Mvu.replaceMvuData(data,{type:'message',message_id:msgId});
      return true;
    }
    return false;
  }catch(err){console.error('Mvu delete failed',err);return false;}
}

                if (!chatMessages || chatMessages.length === 0) return null;
                return chatMessages[0].message;
            } catch (e) {
                return null;
            }
        }

        function parseOptions(messageText) {
            if (!messageText) return [];
            var tagMatch = messageText.match(/<story_options>([\s\S]*?)<\/story_options>/);
            if (!tagMatch || !tagMatch[1]) return [];
            var content = tagMatch[1];
            var regex = /\[([^\|]+)\|([^\]]+)\]/g;
            var match, result = [];
            while ((match = regex.exec(content)) !== null) {
                var typeKey = match[1].trim();
                var desc = match[2].trim();
                var info = TYPE_MAP[typeKey];
                if (info) {
                    result.push({ type: typeKey, css: info.css, icon: info.icon, label: info.label, desc: desc, empty: false });
                }
            }
            return result;
        }

        function escapeHtml(str) {
            var div = document.createElement('div');
            div.textContent = str || '';
            return div.innerHTML;
        }

        function buildOptionsData(parsed) {
            var parsedMap = {};
            for (var i = 0; i < parsed.length; i++) parsedMap[parsed[i].type] = parsed[i];
            optionsData = [];
            for (var j = 0; j < DEFAULTS.length; j++) {
                var type = DEFAULTS[j].type;
                if (parsedMap[type]) {
                    optionsData.push({
                        type: parsedMap[type].type, css: parsedMap[type].css,
                        icon: parsedMap[type].icon, label: parsedMap[type].label,
                        desc: parsedMap[type].desc, empty: false
                    });
                } else {
                    optionsData.push({
                        type: DEFAULTS[j].type, css: DEFAULTS[j].css,
                        icon: DEFAULTS[j].icon, label: DEFAULTS[j].label,
                        desc: DEFAULTS[j].desc, empty: true
                    });
                }
            }
        }

        // ----- 渲染页面 (包含开关UI) -----
        function renderPage() {
            var html = '<div class="main-layout">';
            // 左侧选项区

            for (var i = 0; i < optionsData.length; i++) {
                var opt = optionsData[i];
                var emptyClass = opt.empty ? ' is-empty' : '';
                var delay = 0.05 + i * 0.06;
                html += '<div class="option-item ' + opt.css + emptyClass + ' animate-in" data-index="' + i + '" style="animation-delay:' + delay + 's">';
                html += '<div class="option-icon">' + opt.icon + '</div>';
                html += '<div class="option-text">';
                html += '<div class="option-label">' + escapeHtml(opt.label) + '</div>';
                html += '<span class="option-desc-text">' + escapeHtml(opt.desc) + '</span>';
                html += '<textarea class="option-desc-input">' + escapeHtml(opt.empty ? '' : opt.desc) + '</textarea>';
                html += '</div>';
                html += '<button class="option-edit" data-action="edit" title="编辑选项">';

            html += '<span class="toggle-label">直接发送</span>';
            html += '<div class="toggle-switch"><div class="toggle-knob"></div></div>';

            html += '</div>';

        // ----- 绑定事件 -----
        function bindEvents() {
            var list = document.getElementById('optionsList');
            if (list) {
                list.addEventListener('click', function(e) {
                    var editBtn = e.target.closest('[data-action="edit"]');
                    var item = e.target.closest('.option-item');
                    if (!item) return;
                    var idx = parseInt(item.dataset.index);
                    if (editBtn) {
                        e.stopPropagation();
                        toggleEdit(idx, item);
                        return;
                    }
                    if (e.target.closest('.option-desc-input')) return;
                    if (item.classList.contains('is-editing')) return;
                    if (optionsData[idx].empty) return;
                    handleOptionClick(idx, item);
                });
            }
            var btnSend = document.getElementById('btnSend');
            if (btnSend) {
                btnSend.addEventListener('click', function() {
                    sendCustom();
                });
            }
        }

function handleOptionClick(idx, el) {
    var desc = optionsData[idx].desc;
    if (!desc || desc === '暂无剧情推进') return;
    if (el.classList.contains('is-editing')) return;
    el.classList.add('clicked');
    sendContent(desc);
    setTimeout(function() {
        el.classList.remove('clicked');
    }, 150);
}
        function toggleEdit(idx, item) {
            if (item.classList.contains('is-editing')) {
                var textarea = item.querySelector('.option-desc-input');
                var descText = item.querySelector('.option-desc-text');
                var newVal = (textarea.value || '').trim();
                if (newVal) {
                    optionsData[idx].desc = newVal;
                    optionsData[idx].empty = false;
                    descText.textContent = newVal;
                    item.classList.remove('is-empty');
                } else {
                    optionsData[idx].desc = '暂无剧情推进';
                    optionsData[idx].empty = true;
                    descText.textContent = '暂无剧情推进';
                    item.classList.add('is-empty');
                }
                item.classList.remove('is-editing');
                editingIndex = -1;
            } else {
                var editingItems = document.querySelectorAll('.option-item.is-editing');
                for (var i = 0; i < editingItems.length; i++) {
                    var eidx = parseInt(editingItems[i].dataset.index);
                    toggleEdit(eidx, editingItems[i]);
                }
                var textarea = item.querySelector('.option-desc-input');
                textarea.value = optionsData[idx].empty ? '' : optionsData[idx].desc;
                item.classList.add('is-editing');
                editingIndex = idx;
                setTimeout(function() { textarea.focus(); }, 60);
            }
        }

        function sendCustom() {
            var input = document.getElementById('customInput');
            if (!input) return;
            var val = (input.value || '').trim();
            if (!val) return;
            var btn = document.getElementById('btnSend');
            if (btn) { btn.disabled = true; btn.textContent = '发送中...'; }
            setTimeout(function() {
                sendContent(val);    // 改用统一发送
                if (btn) { btn.disabled = false; btn.textContent = '继续'; }
            }, 180);
        }

        // ----- 开关初始化 (新增) -----
        function initToggle() {
            var toggleContainer = document.getElementById('directSendToggle');
            if (!toggleContainer) return;
            var switchEl = toggleContainer.querySelector('.toggle-switch');
            directSend = loadDirectSend();
            if (directSend) switchEl.classList.add('active');
            else switchEl.classList.remove('active');

            switchEl.onclick = function(e) {
                e.stopPropagation();
                directSend = !directSend;
                if (directSend) switchEl.classList.add('active');
                else switchEl.classList.remove('active');
                saveDirectSend(directSend);
            };
        }

        // ----- 主初始化 -----
        function init() {
            try {
                detectTheme();
                var messageText = getMessageData();
                var parsed = parseOptions(messageText);
                buildOptionsData(parsed);
                renderPage();
            } catch (error) {
                console.error("初始化失败:", error);
                document.getElementById('content').innerHTML = '<div class="loading">加载失败：' + (error.message || error) + '</div>';
            }
        }

        // 启动 (酒馆环境通常有jQuery)
        if (typeof $ !== 'undefined') {
            $(function() { init(); });
        } else {
            window.addEventListener('DOMContentLoaded', init);
        }
    })();
</script>

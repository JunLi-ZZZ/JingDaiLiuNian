function 规范剧情名(text) {
  return String(text || '').trim().replace(/[：:]\s*$/, '');
}

function 解析大纲(raw) {
  const lines = String(raw || '').replace(/\r\n/g, '\n').split('\n');
  const chapters = [];
  let currentStage = null;
  let currentChapter = null;

  for (const rawLine of lines) {
    const line = String(rawLine || '');
    const trimmed = line.trim();
    if (!trimmed) continue;

    const stageMatch = trimmed.match(/^##\s*(.+?)(?:（(.+?)）)?\s*$/);
    if (stageMatch) {
      currentStage = {
        title: stageMatch[1].trim(),
        time: String(stageMatch[2] || '').trim()
      };
      currentChapter = null;
      continue;
    }

    if (!/^\s/.test(line) && /[:：]\s*$/.test(trimmed)) {
      currentChapter = {
        name: 规范剧情名(trimmed),
        stageTitle: currentStage ? currentStage.title : '',
        time: currentStage ? currentStage.time : '',
        fields: []
      };
      chapters.push(currentChapter);
      continue;
    }

    const fieldMatch = line.match(/^\s{2,}([^：:]+)[：:]\s*(.+)\s*$/);
    if (fieldMatch && currentChapter) {
      currentChapter.fields.push({
        key: fieldMatch[1].trim(),
        value: fieldMatch[2].trim()
      });
    }
  }

  return chapters;
}

function 最短路(layerKey, node) {
  const graph = 图缓存[layerKey];
  const dist = new Array(graph.nodes.length).fill(Infinity);
  const prev = new Array(graph.nodes.length).fill(-1);
  const done = new Array(graph.nodes.length).fill(false);
  if (!node) return { graph, dist, prev };

  dist[node.index] = 0;
  for (let step = 0; step < graph.nodes.length; step++) {
    let pick = -1;
    let best = Infinity;
    for (let i = 0; i < dist.length; i++) {
      if (!done[i] && dist[i] < best) {
        best = dist[i];
        pick = i;
      }
    }
    if (pick < 0) break;
    done[pick] = true;
    for (const link of graph.links[pick] || []) {
      const next = dist[pick] + 换算距离(layerKey, link.distance, link.label);
      if (next < dist[link.to]) {
        dist[link.to] = next;
        prev[link.to] = pick;
      }
    }
  }
  return { graph, dist, prev };
}

function 复原路径(prev, startIndex, endIndex) {
  const path = [];
  for (let cur = endIndex; cur >= 0; cur = prev[cur]) {
    path.push(cur);
    if (cur === startIndex) break;
  }
  if (path[path.length - 1] !== startIndex) return null;
  return path.reverse();
}

function 路线经由(graph, path) {

function 五入(value, step) {

}

function 路程分档(candidates, layerKey) {
  const values = candidates.map(route => route.distance).sort((a, b) => a - b);
  if (!values.length) return null;
  const pick = (ratio) => values[Math.max(0, Math.min(values.length - 1, Math.floor((values.length - 1) * ratio)))];
  const fallback = layerKey === 'L2'
    ? { near: 260, cross: 700 }
    : layerKey === 'L3'
      ? { near: 180, cross: 520 }
      : { near: 180, cross: 480 };
  const near = Math.max(fallback.near, 五入(pick(0.3) || fallback.near, 10));
  const cross = Math.max(fallback.cross, near + 50, 五入(pick(0.65) || fallback.cross, 20));
  return { near, cross };
}

function 路线性质(distance, bands) {

  if (distance <= bands.near) return '近距转场';
  if (distance <= bands.cross) return '跨区移动';
  return '长途旅行';
}

function 非相邻参考(layerKey, node, limit = 4) {
  if (!node) return { bands: null, routes: [] };
  const graph = 图缓存[layerKey];
  const { dist, prev } = 最短路(layerKey, node);
  const candidates = [];
  graph.nodes.forEach(target => {
    if (target.index === node.index) return;
    if (!Number.isFinite(dist[target.index]) || dist[target.index] < 15) return;
    if (!是参考节点(graph, target)) return;
    const path = 复原路径(prev, node.index, target.index);
    if (!path || path.length < 3) return;
    candidates.push({
      name: target.name,
      distance: dist[target.index],
      via: 路线经由(graph, path),
      time: 格式耗时(路线耗时(layerKey, graph, path), '', layerKey, dist[target.index]),
      path
    });
  });

  const sorted = candidates
    .sort((a, b) => a.distance - b.distance || a.name.localeCompare(b.name, 'zh-Hans-CN'))
  const bands = 路程分档(sorted, layerKey);
  const picked = [];
  const used = new Set();
  for (const kind of ['近距转场', '跨区移动', '长途旅行']) {
    const found = sorted.find((route, idx) => !used.has(idx) && 路线性质(route.distance, bands) === kind);
    if (!found) continue;
    const idx = sorted.indexOf(found);
    used.add(idx);
    picked.push({ ...found, kind });
  }
  for (let i = 0; i < sorted.length && picked.length < limit; i++) {
    if (used.has(i)) continue;
    picked.push({ ...sorted[i], kind: 路线性质(sorted[i].distance, bands) });
    used.add(i);
  }
  return { bands, routes: picked.slice(0, limit) };
}

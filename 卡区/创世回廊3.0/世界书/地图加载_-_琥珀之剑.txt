function 拆分别名(name) {

function 裁剪地名尾缀(text) {
  const source = String(text || '').trim();
  if (!source) return [];

  const patterns = [
    /(边境|边界|地区|地带|一带|附近|周边|沿岸|近郊|郊外)$/u,
    /(村庄|村)$/u
  ];
  const result = [];
  const seen = new Set([归一化(source)]);
  let current = source;

  while (current) {
    let changed = false;
    for (const pattern of patterns) {
      const next = current.replace(pattern, '').trim();
      const norm = 归一化(next);
      if (!next || next === current || norm.length < 2 || seen.has(norm)) continue;
      seen.add(norm);
      result.push(next);
      current = next;
      changed = true;
      break;

  return result;
}

function 格式耗时(km, mode, speed) {
  if (speed === Infinity) return `${mode}瞬达`;
  if (!Number.isFinite(km) || km <= 0) return `${mode}耗时未知`;
  const days = km / speed;
  if (days <= 0.25) return `${mode}约半日`;
  if (days <= 0.75) return `${mode}约1日内`;
  if (days <= 1.25) return `${mode}约1日`;
  const lo = Math.max(1, Math.floor(days));
  const hi = Math.max(lo, Math.ceil(days));
  return lo === hi ? `${mode}约${lo}日` : `${mode}约${lo}~${hi}日`;
}

          brief: 节点简述(图.nodes[link2.to]),

function 最短路(startNode) {
  const dist = new Array(图.nodes.length).fill(Infinity);
  const prev = new Array(图.nodes.length).fill(-1);
  const done = new Array(图.nodes.length).fill(false);
  if (!startNode) return { dist, prev };

  dist[startNode.index] = 0;
  for (let step = 0; step < 图.nodes.length; step++) {
    let pick = -1;
    let best = Infinity;
    for (let i = 0; i < dist.length; i++) {
      if (!done[i] && dist[i] < best) {
        best = dist[i];
        pick = i;
      }
    }
    if (pick < 0) break;
    done[pick] = true;
    for (const link of 图.links[pick] || []) {
      const next = dist[pick] + 换算距离(link);
      if (next < dist[link.to]) {
        dist[link.to] = next;
        prev[link.to] = pick;
      }
    }
  }
  return { dist, prev };
}

function 复原路径(prev, startIndex, endIndex) {
  const path = [];
  for (let cur = endIndex; cur >= 0; cur = prev[cur]) {
    path.push(cur);
    if (cur === startIndex) break;
  }
  if (path[path.length - 1] !== startIndex) return null;
  return path.reverse();
}

function 路程分档(candidates) {
  const values = candidates.map(route => route.distance).sort((a, b) => a - b);
  if (!values.length) return null;

}

      time: 路线耗时(path)

  });

  const picked = [];
  const used = new Set();

  for (const kind of ['近距转场', '跨区移动', '远程旅行']) {
    const idx = sorted.findIndex((route, routeIndex) => !used.has(routeIndex) && 路线性质(route.distance, bands) === kind);
    if (idx < 0) continue;
    used.add(idx);

  }

    if (used.has(i)) continue;
    used.add(i);
    picked.push({ ...sorted[i], kind: 路线性质(sorted[i].distance, bands) });
  }

  return { bands, routes: picked.slice(0, limit) };
}

    <script>
        // ===== 依赖检测 =====
        function getWin() {
            try { return window.parent || window; } catch (e) { return window; }
        }

        function checkDep(id, test) {
            const dot = document.getElementById('dep-' + id);
            const tag = document.getElementById('dep-' + id + '-tag');
            const ok = test();
            dot.classList.add(ok ? 'ok' : 'fail');
            tag.textContent = ok ? '就绪' : '未检测到';
            tag.classList.add(ok ? 'ok' : 'fail');
            return ok;
        }

        async function runDepChecks() {
            const win = getWin();
            let allOk = true;

            // MVU 变量系统 — 与开始游戏一致，通过 waitGlobalInitialized('Mvu') 检测
            allOk &= checkDep('mvu', () => {
                try {
                    const mvu = win.Mvu;
                    return void 0 !== mvu && typeof mvu.getMvuData === 'function' && typeof mvu.replaceMvuData === 'function';
                } catch (e) { return false; }
            });

            // 辅助计算脚本 — 全局标记
            allOk &= checkDep('calc', () => {
                try {
                    return !!(win.__辅助计算脚本_loaded__);
                } catch (e) { return false; }
            });

            // 小手机脚本 — 全局标记
            allOk &= checkDep('phone', () => {
                try {
                    return !!(win.__小手机脚本_loaded__);
                } catch (e) { return false; }
            });

            // 状态栏.js — 全局标记
            allOk &= checkDep('statusbar', () => {
                try {
                    return !!(win.__状态栏_loaded__);
                } catch (e) { return false; }
            });

            // 提示词模板 (EJS Template) — 与开始游戏一致，通过 SillyTavern extensionSettings 检测
            allOk &= checkDep('prompt', () => {
                try {
                    const top = window.top;
                    if (top && top.SillyTavern) {
                        const ctx = top.SillyTavern.getContext();
                        if (ctx && ctx.extensionSettings) {
                            const ejs = ctx.extensionSettings.EjsTemplate;
                            return !!(ejs && ejs.enabled);
                        }
                    }
                    return false;
                } catch (e) { return false; }
            });

            const summary = document.getElementById('depSummary');
            if (allOk) {
                summary.textContent = '✓ 所有依赖就绪';
                summary.classList.add('all-ok');
            } else {
                summary.textContent = '✗ 部分依赖未就绪，可能影响游戏功能';
                summary.classList.add('has-fail');
            }
        }

        // ===== 图片预加载 + 超时显示 =====
        (function () {
            const loader = document.getElementById('pageLoader');
            const main = document.getElementById('mainContent');
            let revealed = false;

            function showPage() {
                if (revealed) return;
                revealed = true;
                loader.classList.add('fade-out');
                main.classList.add('visible');
                setTimeout(() => loader.remove(), 600);
            }

            // 收集页面中所有 img
            const images = document.querySelectorAll('img');
            if (images.length === 0) {
                showPage();
            } else {
                let loaded = 0;
                const total = images.length;
                images.forEach(img => {
                    if (img.complete) {
                        loaded++;
                        if (loaded >= total) showPage();
                    } else {
                        img.addEventListener('load', () => { if (++loaded >= total) showPage(); });
                        img.addEventListener('error', () => { if (++loaded >= total) showPage(); });
                    }
                });
            }

            // 20秒超时兜底
            setTimeout(showPage, 20000);
        })();

        // 延迟检测，给脚本加载时间
        setTimeout(runDepChecks, 1500);

        // ===== 勾选协议才能点击按钮 =====
        const agreeCheck = document.getElementById('agreeCheck');
        const enterBtn = document.getElementById('enterBtn');

        agreeCheck.addEventListener('change', function () {
            if (this.checked) {
                enterBtn.classList.remove('disabled');
            } else {
                enterBtn.classList.add('disabled');
            }
        });

        enterBtn.addEventListener('click', async function (e) {
            e.preventDefault();
            if (!agreeCheck.checked) return;
            const btn = this;
            btn.classList.add('loading');
            btn.textContent = '正在载入…';

            try {
                const messages = await getChatMessages("0", { include_swipe: true });
                const swipeId = 1;
                if (messages && messages[0].swipes && messages[0].swipes[swipeId]) {
                    await setChatMessage(messages[0].swipes[swipeId], 0, {
                        swipe_id: swipeId,
                        refresh: 'display_and_render_current'
                    });
                    if (typeof toastr !== 'undefined') toastr.success('已进入泰拉深渊');
                } else {
                    throw new Error('开场白2未找到，请确保已添加。');
                }
            } catch (err) {
                console.error('切换失败:', err);
                if (typeof toastr !== 'undefined') toastr.error('进入失败: ' + err.message);
                btn.classList.remove('loading');
                btn.textContent = '踏入深渊';
            }
        });
    </script>

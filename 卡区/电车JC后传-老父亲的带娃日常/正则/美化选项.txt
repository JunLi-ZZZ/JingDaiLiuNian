<script>
  function renderOptions() {
    const container = document.getElementById('dynamic-buttons');
    container.innerHTML = ''; // 清空旧按钮
  
    // 从父页面抓取用户消息
    const userMsg = `$1`;
  
    // 提取所有 <op>…</op>
    const opRegex = /<op>(.*?)<\/op>/gis;
    let match;
    while ((match = opRegex.exec(userMsg)) !== null) {   // ← 这里用 userMsg
      const text = match[1].trim();
      if (!text) continue;
  
      const btn = document.createElement('button');
      btn.className = 'action-btn-silent';
      btn.textContent = text;
      btn.setAttribute('data-action', text);
  
      btn.addEventListener('click', e => {
        e.stopPropagation();
        try {
          const input = window.parent.document.querySelector('#send_textarea');
          if (input) {
            const cur = input.value.trim();
            input.value = cur ? `${cur} ${text}` : text;
            input.dispatchEvent(new Event('input', { bubbles: true }));
          }
        } catch (err) {
          console.error(err);
        }
        btn.classList.toggle('active');
      });
  
      container.appendChild(btn);
    }
  }
  
  // 初始渲染 + 监听父页面输入变化
  renderOptions();
  try {
    const input = window.parent.document.querySelector('#send_textarea');
    if (input) input.addEventListener('input', renderOptions);
  } catch (e) {}
  
  // 展开/收起文字切换
  const details = document.querySelector('.mystic-card-silent');
  const toggleBtn = document.querySelector('.toggle-collapse-btn');
  details?.addEventListener('toggle', () => {
    toggleBtn && (toggleBtn.textContent = details.open ? '收起' : '展开');
  });
</script>
